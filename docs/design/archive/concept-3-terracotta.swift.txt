// Concept 3 — Terracotta, Hand-Thrown
// Warm and tactile: uneven clay shapes, Cochin display over SF Rounded, a

// thumb can set a value fast.

private extension Color {
    init(light: UInt32, dark: UInt32) {
        self.init(UIColor { $0.userInterfaceStyle == .dark ? UIColor(hex: dark) : UIColor(hex: light) })
    }
}

private extension UIColor {
    convenience init(hex: UInt32) {
        self.init(red: CGFloat((hex >> 16) & 0xFF) / 255, green: CGFloat((hex >> 8) & 0xFF) / 255,
                  blue: CGFloat(hex & 0xFF) / 255, alpha: 1)
    }
}

private enum T {
    static let bg = Color(light: 0xF3E3CF, dark: 0x2A1710)
    static let clay = Color(light: 0xE8C7A2, dark: 0x3D2318)
    static let terra = Color(light: 0x962F14, dark: 0xE58A63)
    static let onTerra = Color(light: 0xFFF3E4, dark: 0x2A1710)
    static let ink = Color(light: 0x3A1F14, dark: 0xF7E6D2)
    static let ink2 = Color(light: 0x6B4332, dark: 0xD4B59B)
    static let error = Color(light: 0x9A2A16, dark: 0xFFA58A)

    static func display(_ size: CGFloat, relativeTo style: Font.TextStyle = .title) -> Font {
        .custom("Cochin-Bold", size: size, relativeTo: style)
    }
    static func displayItalic(_ size: CGFloat, relativeTo style: Font.TextStyle = .title) -> Font {
        .custom("Cochin-Italic", size: size, relativeTo: style)
    }
    static func rounded(_ style: Font.TextStyle, _ weight: Font.Weight = .regular) -> Font {
        .system(style, design: .rounded, weight: weight)
    }
}

/// A hand-thrown outline: corners never quite match.
private struct Clay: Shape {
    var tl: CGFloat = 34, tr: CGFloat = 20, bl: CGFloat = 24, br: CGFloat = 40
    func path(in rect: CGRect) -> Path {
        UnevenRoundedRectangle(topLeadingRadius: tl, bottomLeadingRadius: bl, bottomTrailingRadius: br, topTrailingRadius: tr, style: .continuous)
            .path(in: rect)
    }
}

struct ConceptRoot: View {
    var body: some View {
        ConceptShell(tint: T.terra, appearance: configureBars) { model in
            RecipeScreen(model: model)
        } timer: { model in
            TimerScreen(model: model)
        } log: { model in
            LogScreen(model: model)
        } journal: { model in
            JournalScreen(model: model)
        }
    }

    private func configureBars() {
        let tab = UITabBarAppearance()
        tab.configureWithOpaqueBackground()
        tab.backgroundColor = UIColor { $0.userInterfaceStyle == .dark ? UIColor(hex: 0x2A1710) : UIColor(hex: 0xF3E3CF) }
        UITabBar.appearance().standardAppearance = tab
        UITabBar.appearance().scrollEdgeAppearance = tab
        let nav = UINavigationBarAppearance()
        nav.configureWithTransparentBackground()
        let ink = UIColor { $0.userInterfaceStyle == .dark ? UIColor(hex: 0xF7E6D2) : UIColor(hex: 0x3A1F14) }
        nav.largeTitleTextAttributes = [.foregroundColor: ink, .font: UIFont(name: "Cochin-Bold", size: 34) ?? .boldSystemFont(ofSize: 34)]
        nav.titleTextAttributes = [.foregroundColor: ink, .font: UIFont(name: "Cochin-Bold", size: 18) ?? .boldSystemFont(ofSize: 18)]
        UINavigationBar.appearance().standardAppearance = nav
        UINavigationBar.appearance().scrollEdgeAppearance = nav
    }
}

private struct ClayButton: ButtonStyle {
    var filled = true
    @Environment(\.isEnabled) private var enabled
    func makeBody(configuration: Configuration) -> some View {
        configuration.label
            .foregroundStyle(filled ? T.onTerra : T.terra)
            .background(filled ? T.terra : T.clay, in: Clay(tl: 30, tr: 18, bl: 20, br: 34))
            .opacity(enabled ? 1 : 0.4)
            .scaleEffect(configuration.isPressed ? 0.97 : 1)
            .rotationEffect(.degrees(configuration.isPressed ? -0.6 : 0))
            .animation(.spring(duration: 0.22, bounce: 0.35), value: configuration.isPressed)
    }
}

/// Wrapping row of rounded chips.
private struct FlowChips: View {
    let items: [String]
    var body: some View {
        WrapLayout(spacing: 8) {
            ForEach(items, id: \.self) { item in
                Text(item).font(T.rounded(.callout, .semibold)).foregroundStyle(T.ink)
                    .padding(.horizontal, 16).frame(minHeight: 44)
                    .background(T.clay, in: .capsule)
            }
        }
        .accessibilityElement(children: .combine)
    }
}

private struct WrapLayout: Layout {
    var spacing: CGFloat = 8
    func sizeThatFits(proposal: ProposedViewSize, subviews: Subviews, cache: inout ()) -> CGSize {
        let width = proposal.width ?? 320
        var x: CGFloat = 0, y: CGFloat = 0, rowH: CGFloat = 0
        for v in subviews {
            let s = v.sizeThatFits(.unspecified)
            if x + s.width > width { x = 0; y += rowH + spacing; rowH = 0 }
            x += s.width + spacing; rowH = max(rowH, s.height)
        }
        return CGSize(width: width, height: y + rowH)
    }
    func placeSubviews(in bounds: CGRect, proposal: ProposedViewSize, subviews: Subviews, cache: inout ()) {
        var x = bounds.minX, y = bounds.minY, rowH: CGFloat = 0
        for v in subviews {
            let s = v.sizeThatFits(.unspecified)
            if x + s.width > bounds.maxX { x = bounds.minX; y += rowH + spacing; rowH = 0 }
            v.place(at: CGPoint(x: x, y: y), proposal: .unspecified)
            x += s.width + spacing; rowH = max(rowH, s.height)
        }
    }
}

// MARK: - S05 Timer (the wheel)

private struct TimerScreen: View {
    let model: FlowModel
    @Environment(\.dynamicTypeSize) private var dts

    var body: some View {
        TimerClock(model: model) { elapsed, index in
            let r = model.recipe
            let step = r.steps[index]
            let next = model.nextStep(after: index)
            VStack(spacing: 12) {
                HStack {
                    Text(r.name).font(T.displayItalic(18, relativeTo: .body)).foregroundStyle(T.ink2).lineLimit(1)
                    Spacer()
                    Button { model.pauseTimer(); model.showTimer = false } label: {
                        Image(systemName: "xmark").font(.body.weight(.bold)).frame(width: 44, height: 44)
                            .background(T.clay, in: .circle)
                    }
                    .foregroundStyle(T.ink).accessibilityLabel("Close timer")
                }
                dial(elapsed: elapsed, index: index)
                    .frame(maxWidth: dts.isAccessibilitySize ? 230 : 300)
                    .padding(.top, 4)
                ScrollView {
                    VStack(alignment: .leading, spacing: 8) {
                        Text(model.hasStarted ? step.kind.label : "Ready").font(T.display(34, relativeTo: .title)).foregroundStyle(T.terra)
                        Text(model.hasStarted ? step.instruction : "Rinse the filter, add \(Fmt.num(r.doseG)) g of coffee, water at \(Fmt.num(r.tempC))°C.")
                            .font(T.rounded(.title3, .medium)).foregroundStyle(T.ink)
                        if model.hasStarted, let g = step.cumulativeG {
                            Text("Scale reads \(g) g").font(T.rounded(.title3, .bold).monospacedDigit()).foregroundStyle(T.ink)
                        }
                        if model.hasStarted, let next {
                            Text("Then \(next.kind.label) at \(Fmt.clock(next.startS))").font(T.rounded(.subheadline)).foregroundStyle(T.ink2)
                        }
                    }
                    .frame(maxWidth: .infinity, alignment: .leading)
                }
                .scrollBounceBehavior(.basedOnSize)
                controls
            }
            .padding(.horizontal, 20).padding(.bottom, 8)
            .background(T.bg.ignoresSafeArea())
        }
    }

    private func dial(elapsed: TimeInterval, index: Int) -> some View {
        let r = model.recipe
        let frac = min(1, elapsed / Double(r.totalS))
        return ZStack {
            Circle().stroke(T.clay, lineWidth: 22)
            Circle().trim(from: 0, to: frac).stroke(T.terra, style: StrokeStyle(lineWidth: 22, lineCap: .round))
                .rotationEffect(.degrees(-90))
            ForEach(r.steps) { s in
                Circle().fill(Double(s.startS) <= elapsed ? T.onTerra : T.terra).frame(width: 7, height: 7)
                    .offset(y: -0) // positioned by rotation below
                    .offset(x: 0, y: 0)
                    .modifier(OnRing(angle: Double(s.startS) / Double(r.totalS) * 360))
            }
            VStack(spacing: 0) {
                Text(Fmt.clock(Int(elapsed))).font(T.display(64, relativeTo: .largeTitle)).monospacedDigit()
                    .dynamicTypeSize(...DynamicTypeSize.xxxLarge)
                    .minimumScaleFactor(0.5).lineLimit(1).foregroundStyle(T.ink)
                Text("of \(Fmt.clock(r.totalS))").font(T.rounded(.subheadline)).foregroundStyle(T.ink2)
                    .dynamicTypeSize(...DynamicTypeSize.large).lineLimit(1)
            }
            .padding(34)
        }
        .padding(11)
        .aspectRatio(1, contentMode: .fit)
        .accessibilityElement()
        .accessibilityLabel("Elapsed \(Int(elapsed) / 60) minutes \(Int(elapsed) % 60) seconds, step \(index + 1) of \(r.steps.count)")
    }

    private var controls: some View {
        HStack(spacing: 12) {
            if model.hasStarted && !model.isDone {
                Button { model.finishBrew() } label: {
                    Text("End and log").font(T.rounded(.headline, .bold)).dynamicTypeSize(...DynamicTypeSize.accessibility1)
                        .frame(maxWidth: .infinity, minHeight: 60)
                }
                .buttonStyle(ClayButton(filled: false))
            }
            Button { model.isDone ? model.finishBrew() : model.toggleTimer() } label: {
                Text(model.isDone ? "Log it" : (!model.hasStarted ? "Start" : (model.isRunning ? "Pause" : "Resume")))
                    .font(T.display(24, relativeTo: .title2)).dynamicTypeSize(...DynamicTypeSize.accessibility1)
                    .frame(maxWidth: .infinity, minHeight: 60)
            }
            .buttonStyle(ClayButton())
        }
    }
}

private struct OnRing: ViewModifier {
    let angle: Double
    func body(content: Content) -> some View {
        GeometryReader { g in
            content.position(x: g.size.width / 2 + (g.size.width / 2 - 11) * CGFloat(sin(angle * .pi / 180)),
                             y: g.size.height / 2 - (g.size.height / 2 - 11) * CGFloat(cos(angle * .pi / 180)))
        }
    }
}

                        .font(T.rounded(.subheadline)).foregroundStyle(T.ink2)
                    slider("Coffee", "g", $model.draft.doseG, 1...40, 0.5, target: model.recipe.doseG)
                    slider("Water", "g", $model.draft.waterG, 20...600, 5, target: model.recipe.waterG)
                    slider("Temperature", "°C", $model.draft.tempC, 70...100, 1, target: model.recipe.tempC)
                    slider("Time", "s", Binding(get: { Double(model.draft.timeS) }, set: { model.draft.timeS = Int($0) }),

                    slider("Grind", "", Binding(get: { Double(model.draft.grind) }, set: { model.draft.grind = Int($0) }),

                    if let ratio = model.draft.ratio {
                        Text("Ratio 1:\(Fmt.num((ratio * 10).rounded() / 10))").font(T.displayItalic(22, relativeTo: .title3)).foregroundStyle(T.terra)
                    }
                    Text("How did it taste?").font(T.display(20, relativeTo: .title3)).foregroundStyle(T.ink)
                    LazyVGrid(columns: [GridItem(.adaptive(minimum: 104), spacing: 8)], spacing: 8) {
                        ForEach(Verdict.allCases) { v in
                            let on = model.draft.verdict == v
                            Button { model.draft.verdict = on ? nil : v } label: {
                                Text(v.label).font(T.rounded(.callout, .semibold))
                                    .frame(maxWidth: .infinity, minHeight: 48)
                                    .foregroundStyle(on ? T.onTerra : T.ink)
                                    .background(on ? T.terra : T.clay, in: .capsule)
                            }
                            .accessibilityAddTraits(on ? .isSelected : [])
                        }
                    }
                    Text("Grade  \(Fmt.stars(model.draft.ratingHalfSteps))").font(T.display(20, relativeTo: .title3)).foregroundStyle(T.ink)
                    mugs
                    Text("Notes").font(T.display(20, relativeTo: .title3)).foregroundStyle(T.ink)
                    TextField("What you tasted, what to change", text: $model.draft.notes, axis: .vertical)
                        .font(T.rounded(.body)).lineLimit(3...6).padding(16)
                        .background(T.clay, in: Clay(tl: 22, tr: 30, bl: 30, br: 20))
                    if let err = model.saveError { Text(err).font(T.rounded(.callout, .semibold)).foregroundStyle(T.error) }
                }
                .padding(.horizontal, 20).padding(.top, 6).padding(.bottom, 20)
            }
            .scrollDismissesKeyboard(.interactively)
            .background(T.bg)
            .navigationTitle(model.draft.bean)
            .navigationBarTitleDisplayMode(.inline)
            .toolbar { ToolbarItem(placement: .cancellationAction) { Button("Cancel") { dismiss() } } }
            .safeAreaInset(edge: .bottom) {
                Button { model.save() } label: {
                    Text("Save brew").font(T.display(22, relativeTo: .title3)).dynamicTypeSize(...DynamicTypeSize.accessibility1)
                        .frame(maxWidth: .infinity, minHeight: 60)
                }
                .buttonStyle(ClayButton()).disabled(!model.draft.canSave)
                .padding(.horizontal, 20).padding(.vertical, 8).background(T.bg)
            }
        }
    }

    private func slider(_ title: String, _ unit: String, _ value: Binding<Double>, _ range: ClosedRange<Double>,
                        _ step: Double, target: Double?) -> some View {
        VStack(alignment: .leading, spacing: 4) {
            HStack(alignment: .firstTextBaseline) {
                Text(title).font(T.display(19, relativeTo: .headline)).foregroundStyle(T.ink)
                Spacer()
                Text(unit == "s" ? Fmt.clock(Int(value.wrappedValue)) : "\(Fmt.num(value.wrappedValue)) \(unit)")
                    .font(T.display(26, relativeTo: .title2)).monospacedDigit().foregroundStyle(T.terra)
            }
            Slider(value: value, in: range, step: step)
                .tint(T.terra)
                .frame(minHeight: 44)
                .accessibilityLabel(title)
                .accessibilityValue(unit == "s" ? Fmt.clock(Int(value.wrappedValue)) : "\(Fmt.num(value.wrappedValue)) \(unit == "g" ? "grams" : unit)")
            if let target {
                Text("Recipe target \(unit == "s" ? Fmt.clock(Int(target)) : Fmt.num(target) + " " + unit)")
                    .font(T.rounded(.caption)).foregroundStyle(T.ink2)
            }
        }
    }

    /// Five mugs: tap for whole, tap again for half, again to clear.
    private var mugs: some View {
        HStack(spacing: 6) {
            ForEach(1...5, id: \.self) { i in
                let half = model.draft.ratingHalfSteps
                Button {
                    let whole = i * 2
                    model.draft.ratingHalfSteps = half == whole ? whole - 1 : (half == whole - 1 ? 0 : whole)
                } label: {
                    Image(systemName: "cup.and.saucer.fill").font(.system(size: 30))
                        .foregroundStyle(half >= i * 2 ? T.terra : (half == i * 2 - 1 ? T.terra.opacity(0.55) : T.clay))
                        .frame(maxWidth: .infinity, minHeight: 56)
                        .background(T.clay.opacity(0.45), in: Clay(tl: 18, tr: 12, bl: 14, br: 20))
                }
                .accessibilityHidden(true)
            }
        }
        .accessibilityElement()
        .accessibilityLabel("Grade")
        .accessibilityValue(Fmt.stars(model.draft.ratingHalfSteps))
        .accessibilityAdjustableAction { dir in
            let h = model.draft.ratingHalfSteps
            model.draft.ratingHalfSteps = dir == .increment ? min(10, h + 1) : max(0, h - 1)
        }
    }
}
